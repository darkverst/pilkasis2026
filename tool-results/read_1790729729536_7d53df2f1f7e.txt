    79→/* ============================== LOGIN ============================== */
    80→function AdminLogin({ onSuccess }: { onSuccess: () => void }) {
    81→  const [password, setPassword] = useState("");
    82→  const [loading, setLoading] = useState(false);
    83→  const [error, setError] = useState<string | null>(null);
    84→  const { toast } = useToast();
    85→
    86→  const submit = async () => {
    87→    if (!password) return;
    88→    setLoading(true);
    89→    setError(null);
    90→    try {
    91→      const res = await fetch("/api/admin/login", {
    92→        method: "POST",
    93→        headers: { "Content-Type": "application/json" },
    94→        body: JSON.stringify({ password }),
    95→      });
    96→      const data = await res.json();
    97→      if (!res.ok) {
    98→        setError(data.error || "Login gagal.");
    99→        return;
   100→      }
   101→      toast({ title: "Selamat datang, Panitia!", description: "Anda berhasil masuk." });
   102→      onSuccess();
   103→    } catch {
   104→      setError("Koneksi bermasalah.");
   105→    } finally {
   106→      setLoading(false);
   107→    }
   108→  };
   109→
   110→  return (
   111→    <div className="mx-auto max-w-md">
   112→      <Card className="border-blue-100 bg-white/85 shadow-xl shadow-blue-100/40">
   113→        <CardContent className="p-6 sm:p-8">
   114→          <div className="text-center">
   115→            <div className="mx-auto mb-4 flex h-14 w-14 items-center justify-center rounded-2xl bg-blue-100 text-blue-600">
   116→              <Lock className="h-7 w-7" />
   117→            </div>
   118→            <h2 className="text-2xl font-black text-blue-950">Login Panitia</h2>
   119→            <p className="mt-2 text-sm text-blue-700/80">
   120→              Masukkan password panitia untuk mengelola pemilihan OSIS.
   121→            </p>
   122→          </div>
   123→          <div className="mt-6 space-y-3">
   124→            <Label htmlFor="pw" className="text-blue-800">Password Panitia</Label>
   125→            <Input
   126→              id="pw"
   127→              type="password"
   128→              value={password}
   129→              onChange={(e) => { setPassword(e.target.value); setError(null); }}
   130→              onKeyDown={(e) => e.key === "Enter" && submit()}
   131→              placeholder="••••••••"
   132→              className="border-blue-200 focus:border-blue-500"
   133→              autoFocus
   134→            />
   135→            {error && <p className="text-sm text-rose-600">{error}</p>}
   136→            <Button
   137→              onClick={submit}
   138→              disabled={loading || !password}
   139→              className="w-full bg-blue-600 text-white shadow-lg shadow-blue-600/30 hover:bg-blue-700"
   140→              size="lg"
   141→            >
   142→              {loading ? <Loader2 className="mr-2 h-4 w-4 animate-spin" /> : <ShieldCheck className="mr-2 h-4 w-4" />}
   143→              Masuk
   144→            </Button>
   145→            <p className="rounded-lg bg-blue-50 p-3 text-center text-xs text-blue-700/80">
   146→              Default: <span className="font-mono font-bold">panitia2025</span> (atur via env{" "}
   147→              <span className="font-mono">ADMIN_PASSWORD</span>)
   148→            </p>
   149→          </div>
   150→        </CardContent>
   151→      </Card>
   152→    </div>
   153→  );
   154→}

   298→function CandidatesTab() {
   299→  const [items, setItems] = useState<Candidate[]>([]);
   300→  const [loading, setLoading] = useState(true);
   301→  const [editing, setEditing] = useState<Candidate | null>(null);
   302→  const [creating, setCreating] = useState(false);
   303→  const { toast } = useToast();
   304→
   305→  const load = (showLoading = false) => {
   306→    if (showLoading) setLoading(true);
   307→    fetch("/api/candidates").then((r) => r.json()).then((d: Candidate[]) => setItems(d || [])).finally(() => setLoading(false));
   308→  };
   309→  useEffect(() => {
   310→    let alive = true;
   311→    fetch("/api/candidates")
   312→      .then((r) => r.json())
   313→      .then((d: Candidate[]) => alive && setItems(d || []))
   314→      .finally(() => alive && setLoading(false));
   315→    return () => { alive = false; };
   316→  }, []);
   317→
   318→  const handleDelete = async (c: Candidate) => {
   319→    if (!confirm(`Hapus calon "${c.name}"? Semua suara terkait juga akan dihapus.`)) return;
   320→    const res = await fetch(`/api/admin/candidates/${c.id}`, { method: "DELETE" });
   321→    if (res.ok) {
   322→      toast({ title: "Calon dihapus." });
   323→      load();
   324→    } else {
   325→      toast({ title: "Gagal menghapus.", variant: "destructive" });
   326→    }
   327→  };
   328→
   329→  return (
   330→    <div className="space-y-4">
   331→      <div className="flex items-center justify-between">
   332→        <p className="text-sm text-blue-700/80">{items.length} calon terdaftar</p>
   333→        <Button onClick={() => setCreating(true)} className="bg-blue-600 text-white hover:bg-blue-700">
   334→          <Plus className="mr-1.5 h-4 w-4" /> Tambah Calon
   335→        </Button>
   336→      </div>
   337→
   338→      {loading ? (
   339→        <div className="py-10 text-center"><Loader2 className="mx-auto h-6 w-6 animate-spin text-blue-400" /></div>
   340→      ) : items.length === 0 ? (
   341→        <Card className="border-blue-100 bg-white/80"><CardContent className="p-8 text-center text-blue-700/70">Belum ada calon. Tambahkan calon pertama Anda.</CardContent></Card>
   342→      ) : (
   343→        <div className="grid grid-cols-1 gap-3 sm:grid-cols-2 lg:grid-cols-3">
   344→          {items.map((c, i) => (
   345→            <Card key={c.id} className="overflow-hidden border-blue-100 bg-white/85">
   346→              <div className="flex">
   347→                <div className="h-24 w-24 shrink-0 overflow-hidden bg-blue-50">
   348→                  {c.photo ? (
   349→                     
   350→                    <img src={c.photo} alt={c.name} className="h-full w-full object-cover" />
   351→                  ) : (
   352→                    <div className="flex h-full w-full items-center justify-center text-3xl font-black text-blue-200">{c.name.charAt(0)}</div>
   353→                  )}
   354→                </div>
   355→                <div className="flex-1 p-3">
   356→                  <div className="flex items-start justify-between gap-2">
   357→                    <div className="min-w-0">
   358→                      <p className="text-xs text-blue-500">No. Urut {c.order || i + 1}</p>
   359→                      <p className="truncate font-bold text-blue-950">{c.name}</p>
   360→                      <p className="text-xs text-blue-600">{c.class}</p>
   361→                    </div>
   362→                    <span className="h-3 w-3 shrink-0 rounded-full" style={{ backgroundColor: c.color }} />
   363→                  </div>
   364→                  <p className="mt-1 line-clamp-2 text-xs text-blue-700/70">{c.vision}</p>
   365→                  <div className="mt-2 flex gap-2">
   366→                    <Button size="sm" variant="outline" onClick={() => setEditing(c)} className="border-blue-200 text-blue-700 hover:bg-blue-50 h-7 px-2 text-xs">
   367→                      <Pencil className="mr-1 h-3 w-3" /> Edit
   368→                    </Button>

   370→                      <Trash2 className="mr-1 h-3 w-3" /> Hapus
   371→                    </Button>
   372→                  </div>
   373→                </div>
   374→              </div>
   375→            </Card>
   376→          ))}
   377→        </div>
   378→      )}
   379→
   380→      <CandidateFormDialog
   381→        key={creating ? "create" : "create-closed"}
   382→        open={creating}
   383→        candidate={null}
   384→        onClose={() => setCreating(false)}
   385→        onSaved={() => { setCreating(false); load(); }}
   386→      />
   387→      <CandidateFormDialog
   388→        key={editing?.id || "edit-closed"}
   389→        open={!!editing}
   390→        candidate={editing}
   391→        onClose={() => setEditing(null)}
   392→        onSaved={() => { setEditing(null); load(); }}
   393→      />
   394→    </div>
   395→  );
   396→}

   398→function CandidateFormDialog({
   399→  open,
   400→  candidate,
   401→  onClose,
   402→  onSaved,
   403→}: {
   404→  open: boolean;
   405→  candidate: Candidate | null;
   406→  onClose: () => void;
   407→  onSaved: () => void;
   408→}) {
   409→  const { toast } = useToast();
   410→  // Initialize state directly from the candidate prop. The parent passes a `key`
   411→  // so the component remounts fresh each time it opens for a different candidate,
   412→  // avoiding prop-to-state sync inside an effect.
   413→  const [name, setName] = useState(candidate?.name || "");
   414→  const [kelas, setKelas] = useState(candidate?.class || "");
   415→  const [vision, setVision] = useState(candidate?.vision || "");
   416→  const [mission, setMission] = useState(candidate?.mission || "");
   417→  const [order, setOrder] = useState(String(candidate?.order ?? 1));
   418→  const [color, setColor] = useState(candidate?.color || "#3b82f6");
   419→  const [photo, setPhoto] = useState(candidate?.photo || "");
   420→  const [saving, setSaving] = useState(false);
   421→  const fileRef = useRef<HTMLInputElement>(null);
   422→
   423→  const handlePhoto = (file: File) => {
   424→    if (file.size > 2_500_000) {
   425→      toast({ title: "Ukuran foto terlalu besar", description: "Maksimal 2.5MB.", variant: "destructive" });
   426→      return;
   427→    }
   428→    const reader = new FileReader();
   429→    reader.onload = () => setPhoto(String(reader.result));
   430→    reader.readAsDataURL(file);
   431→  };
   432→
   433→  const save = async () => {
   434→    if (!name.trim() || !kelas.trim() || !vision.trim() || !mission.trim()) {
   435→      toast({ title: "Lengkapi semua field wajib.", variant: "destructive" });
   436→      return;
   437→    }
   438→    setSaving(true);
   439→    const body = { name, class: kelas, vision, mission, order: Number(order) || 1, color, photo };
   440→    const url = candidate ? `/api/admin/candidates/${candidate.id}` : "/api/admin/candidates";
   441→    const method = candidate ? "PUT" : "POST";
   442→    const res = await fetch(url, { method, headers: { "Content-Type": "application/json" }, body: JSON.stringify(body) });
   443→    setSaving(false);
   444→    if (res.ok) {
   445→      toast({ title: candidate ? "Calon diperbarui." : "Calon ditambahkan." });
   446→      onSaved();
   447→    } else {
   448→      toast({ title: "Gagal menyimpan.", variant: "destructive" });
   449→    }
   450→  };
   451→
   452→  const colors = ["#3b82f6", "#0ea5e9", "#06b6d4", "#6366f1", "#14b8a6", "#8b5cf6", "#ec4899", "#f59e0b"];
   453→
   454→  return (
   455→    <Dialog open={open} onOpenChange={(o) => !o && onClose()}>
   456→      <DialogContent className="max-h-[90vh] overflow-y-auto scrollbar-blue sm:max-w-lg">
   457→        <DialogHeader>
   458→          <DialogTitle className="text-blue-950">{candidate ? "Edit Calon" : "Tambah Calon"}</DialogTitle>
   459→          <DialogDescription>Lengkapi identitas dan visi-misi calon OSIS.</DialogDescription>
   460→        </DialogHeader>
   461→        <div className="space-y-3">
   462→          {/* Photo */}
   463→          <div className="flex items-center gap-3">
   464→            <div className="h-20 w-20 shrink-0 overflow-hidden rounded-xl bg-blue-50 ring-1 ring-blue-100">
   465→              {photo ? (
   466→                 
   467→                <img src={photo} alt="preview" className="h-full w-full object-cover" />
   468→              ) : (
   469→                <div className="flex h-full w-full items-center justify-center text-blue-300"><GraduationCap className="h-7 w-7" /></div>
   470→              )}
   471→            </div>
   472→            <div className="flex-1 space-y-1">
