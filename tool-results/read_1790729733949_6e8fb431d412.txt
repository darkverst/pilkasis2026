   398→   398→function CandidateFormDialog({

   402→   402→  onSaved,
   403→   403→}: {
   404→   404→  open: boolean;

   407→   407→  onSaved: () => void;
   408→   408→}) {
   409→   409→  const { toast } = useToast();
   410→   410→  // Initialize state directly from the candidate prop. The parent passes a `key`
   411→   411→  // so the component remounts fresh each time it opens for a different candidate,
   412→   412→  // avoiding prop-to-state sync inside an effect.
